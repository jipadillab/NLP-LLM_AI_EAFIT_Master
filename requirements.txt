# ============================================================
# NLP & LLM Interactive Lab — EAFIT
# requirements.txt  (SIN PyTorch / torchvision / transformers)
# ============================================================
# Instalación local:  pip install -r requirements.txt
# Streamlit Cloud:     sube main_app.py + requirements.txt + .streamlit/config.toml
#
# POR QUÉ NO HAY torch / torchvision / transformers / sentence-transformers:
# `transformers` registra internamente cientos de submódulos (incluye
# modelos de visión como ViTMatte/ViTPose/YOLOS), y el *file watcher* de
# Streamlit los recorre a todos al arrancar, disparando
# `ModuleNotFoundError: No module named 'torchvision'` en los logs aunque
# la app nunca los use. La única forma robusta de eliminar ese error es
# no depender de esas librerías. Aquí se reemplazan por alternativas
# 100% CPU, SIN PyTorch:
#   • tiktoken   → tokenización BPE (GPT-2 original y cl100k moderno)
#   • tokenizers → tokenización WordPiece de BERT (Rust puro)
#   • fastembed  → embeddings densos vía ONNX Runtime
# ============================================================

# --- Core app ---
streamlit>=1.38,<2.0
pandas>=2.0,<3.0
numpy>=1.26,<2.0

# --- Visualización ---
plotly>=5.20,<6.0

# --- Groq API (todos los modelos de generación de este lab, gratis) ---
groq>=0.11,<1.0

# --- Tokenización (sin PyTorch) ---
tiktoken>=0.7,<1.0            # BPE: cl100k (ref. GPT-3.5/4) y "gpt2" (BPE original, 2019)
tokenizers>=0.19,<1.0         # WordPiece de BERT — Rust puro, CERO dependencia de PyTorch

# --- NLP clásico ---
nltk>=3.9,<4.0                 # POS, NER, VADER sentiment, n-gramas

# --- Embeddings, TF-IDF, similitud coseno y proyecciones ---
scikit-learn>=1.4,<2.0         # TF-IDF, cosine_similarity, TruncatedSVD (PCA-like)
fastembed>=0.3,<1.0            # all-MiniLM-L6-v2 / bge-m3 / nomic-embed-text vía ONNX Runtime
